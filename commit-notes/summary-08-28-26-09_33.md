## Overview

This commit implements the event-driven S3 ingestion Lambda function (`lambda/ingestion/handler.py`) and its automated Terraform infrastructure (`terraform/lambda_ingestion.tf`). The Lambda service automatically triggers when new transaction CSVs land under `raw/*.csv` in the S3 bucket, streaming records row-by-row to maintain low memory consumption. It executes a hybrid fraud scoring engine combining deterministic business rules (high monetary amount, high risk score with international origin) and machine learning inference (`model.joblib`), writing all flagged anomalies in batches to DynamoDB with structured CloudWatch logging.

## Code Changes

### Hybrid Fraud Scoring Engine (Rules + ML Inference)
In `lambda/ingestion/handler.py`, transaction records are evaluated against both business heuristic rules and machine learning model predictions, assigning detection categories:
```python
def evaluate_transaction(row: Dict[str, Any]) -> Optional[Dict[str, Any]]:
    """Evaluate a single transaction record against hybrid fraud detection logic."""
    tx_id = row.get("TransactionID") or row.get("transaction_id", "")
    acc_id = row.get("AccountID") or row.get("account_id", "")
    ts = row.get("Timestamp") or row.get("timestamp", "")

    if not tx_id or not acc_id:
        return None

    # Parse numerical features
    amount = parse_numeric(row.get("Amount") or row.get("amount", 0.0))
    tx_freq = parse_int(row.get("TransactionFrequency") or row.get("transaction_frequency", 1))
    distance = parse_numeric(row.get("DistanceFromLastTx") or row.get("distance_from_last_tx", 0.0))
    is_intl = parse_int(row.get("IsInternational") or row.get("is_international", 0))
    risk_score = parse_numeric(row.get("RiskScore") or row.get("risk_score", 0.0))

    # 1. Rule-based evaluation ($5,000 threshold or high risk + international)
    rule_flagged, rule_reasons = evaluate_rules(
        amount=amount,
        risk_score=risk_score,
        is_international=is_intl,
    )

    # 2. Machine Learning evaluation (RandomForest model inference)
    feature_vector = [amount, float(tx_freq), distance, float(is_intl), risk_score]
    ml_flagged, ml_prob, ml_reason = evaluate_ml(feature_vector)

    # 3. Determine detection type and combined reason
    if rule_flagged and ml_flagged:
        detection_type = "HYBRID"
        combined_reasons = rule_reasons + ([ml_reason] if ml_reason else [])
        flag_reason = " | ".join(combined_reasons)
    elif rule_flagged:
        detection_type = "RULE"
        flag_reason = " | ".join(rule_reasons)
    elif ml_flagged:
        detection_type = "ML"
        flag_reason = ml_reason or "ML Model detected anomalous fraud pattern"
    else:
        return None  # Legitimate transaction
```

### Memory-Efficient S3 Streaming and DynamoDB Batch Ingestion
In `lambda/ingestion/handler.py`, incoming S3 object bodies are streamed line-by-line and flushed to DynamoDB using `batch_writer()` to avoid memory bloat on large batches:
```python
def process_s3_file(bucket_name: str, object_key: str) -> Dict[str, Any]:
    """Stream and parse S3 CSV row-by-row and write flagged items to DynamoDB."""
    s3_response = s3_client.get_object(Bucket=bucket_name, Key=object_key)
    body_stream = s3_response["Body"]

    def line_generator():
        for line in body_stream.iter_lines():
            if line:
                yield line.decode("utf-8")

    csv_reader = csv.DictReader(line_generator())
    flagged_batch: List[Dict[str, Any]] = []

    for row in csv_reader:
        records_scanned += 1
        flagged_record = evaluate_transaction(row)

        if flagged_record is not None:
            records_flagged += 1
            flagged_batch.append(flagged_record)

            if len(flagged_batch) >= 25:
                write_flagged_batch(flagged_batch)
                flagged_batch = []

    if flagged_batch:
        write_flagged_batch(flagged_batch)
```

### Automated Lambda Packaging & S3 Notification Trigger
In `terraform/lambda_ingestion.tf`, the function code and pre-trained model are automatically zipped and wired to S3 `ObjectCreated` event notifications for `raw/*.csv`:
```hcl
data "archive_file" "ingestion_lambda_zip" {
  type        = "zip"
  source_dir  = "${path.module}/../lambda/ingestion"
  output_path = "${path.module}/build/ingestion_lambda.zip"
}

resource "aws_lambda_function" "ingestion_lambda" {
  function_name    = "${var.project_name}-ingestion"
  filename         = data.archive_file.ingestion_lambda_zip.output_path
  source_code_hash = data.archive_file.ingestion_lambda_zip.output_base64sha256
  runtime          = "python3.11"
  handler          = "handler.lambda_handler"
  memory_size      = 512
  timeout          = 60
  role             = aws_iam_role.ingestion_lambda_role.arn

  environment {
    variables = {
      DYNAMODB_TABLE_NAME = aws_dynamodb_table.flagged_transactions.name
      MODEL_PATH          = "model.joblib"
      AWS_REGION_NAME     = var.aws_region
    }
  }
}

resource "aws_s3_bucket_notification" "raw_transactions_notification" {
  bucket = aws_s3_bucket.raw_transactions.id

  lambda_function {
    lambda_function_arn = aws_lambda_function.ingestion_lambda.arn
    events              = ["s3:ObjectCreated:*"]
    filter_prefix       = "raw/"
    filter_suffix       = ".csv"
  }

  depends_on = [
    aws_lambda_permission.allow_s3_ingestion,
  ]
}
```
